        /* =====================================================
           VALIDAÇÃO DO LOGIN

        async function validarAcesso() {


            const user =
                document
                .getElementById("usuario")
                .value
                .trim();


            const senha =
                document
                .getElementById("senha")
                .value;


            const perfil =
                document
                .getElementById("perfil_logado")
                .value;


            /* Verifica se os campos foram preenchidos */

            if (!user || !senha) {

                alert(
                    "Preencha os campos para continuar!"
                );

                return;
            }


            try {


                const response =
                    await fetch(
                        "/login",
                        {

                            method: "POST",

                            headers: {
                                "Content-Type":
                                    "application/json"
                            },

                            body: JSON.stringify({

                                user: user,

                                senha: senha,

                                tipo: perfil

                            })

                        }
                    );


                const dados =
                    await response.json();


                /* LOGIN REALIZADO */

                if (dados.success) {


                    alert(
                        "Conexão estabelecida! Bem-vindo, "
                        + dados.nome
                    );


                    /*
                     * O servidor informa
                     * para qual página
                     * o usuário deve ir.
                     */

                    window.location.href =
                        dados.redirect;


                } else {


                    /* LOGIN NEGADO */

                    alert(
                        "❌ Erro: "
                        + dados.message
                    );

                }


            } catch (error) {


                console.error(error);


                alert(
                    "Servidor offline. Verifique o terminal!"
                );

            }

        }

    </script>
